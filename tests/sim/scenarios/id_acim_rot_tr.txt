# idacim rotating tr check against a plant whose tr (0.12) is 1.4x the
# standstill estimate's region, on the spindle's FOC config.
title idacim rotating tr check
config conf/fanuc_a2_spindle_foc_enc.txt
f4 link id_acim
f4 fault0.en = 0
f4 idacim0.test_cur = 16
f4 idacim0.n_volt = 155
f4 idacim0.n_freq = 78.4
f4 idacim0.n_cur = 19
f4 idacim0.n_pp = 2
f4 idacim0.spin = 1
f4 idacim0.rot_tr_vel = 600
plant r=0.19 ld=0.0008 lmr=0.0124 tr=0.12 i_knee=0 lmr_sat=0 tr_sat=0 i_dip=0 lmr_dip=0 tr_dip=0 jm=0.0118 f=0.5 d=0.0013 dt_shape=1 dt_i0=0.85 dt_eff=1.77e-6
at 0.1 f4 fault0.en = 1
at 0.2 f4 idacim0.state = 1.2
at 21 f4 idacim0.state = 4.2
trace idacim0.state hv0.id_fb hv0.iq_fb plant.vel plant.i_mr idacim0.d_cmd idacim0.q_cmd
expect max fault0.last_fault < 0.5 from 0.15
expect final idacim0.tr > 0
expect final idacim0.rot_tr > 0
expect final idacim0.rot_tr_k > 0
expect final idacim0.rot_tr_imr > 0
expect final idacim0.rot_tr_lmr > 0
expect final idacim0.rot_tr_b > -1
run 80
