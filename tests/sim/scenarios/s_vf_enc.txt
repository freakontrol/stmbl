# Spindle V/f with encoder slip control: 0 -> 838 rad/s and back. The
# plant carries the spindle's identified flux shape (the config has none).
title Spindle V/f encoder, 0 -> 838 -> 0 rad/s
config conf/fanuc_a2_spindle_vf_enc.txt
plant lmr=0.012432 tr=0.090593 i_n=19.48 i_knee=14.4 lmr_sat=0.021708 tr_sat=0.021708 i_dip=10.48 lmr_dip=0.3488 tr_dip=-0.45 f=0.499 d=0.00128
f4 vf0.vel_cmd = 0
at 0.1 f4 fault0.en = 1
at 0.2 f4 vf0.vel_cmd = 838
at 10.0 f4 vf0.vel_cmd = 0
trace plant.vel vf0.vel vf0.stall hv0.id_fb hv0.iq_fb plant.i_mr vf0.u_cmd vf0.slip
expect max fault0.last_fault < 0.5 from 0.15
expect final plant.vel > 825 from 9.9 to 10.0
expect max hv0.iq_fb < 30
# Sim finding (10 Oct, not seen on the bench yet): braking below ~25 rad/s the
# boost fades on the rotor speed while the field runs slip_max below it, so
# u_cmd drops to ~5 V, the flux collapses (i_mr 19 -> 1 A) and the rotor
# hunts +-15 rad/s around zero for ~2 s before it settles.
expect absmax plant.vel < 16 from 18.6 to 21
expect absmax plant.vel < 1 from 21.5
run 22.5
