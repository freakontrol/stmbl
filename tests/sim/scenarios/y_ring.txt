# The same move with the screw as a second mass. ring_hz is the free
# two-mass resonance: Y's FRF (12 and 15 Oct) has the peak at 26-29 Hz and
# the anti-resonance near 14 Hz. With the config's inertias (j_sys/j 1.42)
# the model can not place both; 27.5 Hz keeps the peak, the dip lands near
# 17.6 Hz. zeta 0.2. pid0 filters the inertia feedforward at conf0.j_lpf.
title Y move with the load ring (27.5 Hz free resonance)
config conf/fanuc_a3-3000.txt
f4 pid0.hold_en = 0
# Y friction is the plant's: the config runs dob with friction ff 0
plant f=0.526805 d=0.010228
plant ring_hz=27.5 ring_zeta=0.2
lcnc scale 10
at 0.05 lcnc enable 1
at 0.1 lcnc move 20 200 1500
trace lcnc.pos plant.pos_mm plant.vel plant.load_vel plant.twist plant.shaft pid0.pos_error pid0.torque_cmd hv0.iq_fb
expect max fault0.last_fault < 0.5 from 0.06
expect absmax pid0.pos_error < 0.2
# settled: no ring left (friction may hold some twist in the screw)
expect std plant.twist < 0.002 from 0.5
expect absmax plant.load_vel < 0.05 from 0.5
expect final linrev0.fb_out > 19.99
expect final linrev0.fb_out < 20.01
expect final plant.pos_mm > 19.9
run 0.7
