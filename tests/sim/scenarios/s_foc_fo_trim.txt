# Spindle FOC on the encoder with the flux_obs slip trim, plant tr 1.3x the
# config's (a cold rotor): k = tr_model / tr_true = 0.771, and
# the frame stay on the flux.
title flux_obs trim, plant tr 1.3x the model, 600 rad/s
config conf/fanuc_a2_spindle_foc_enc.txt
plant tr=0.1314
f4 link flux_obs
f4 flux_obs0.kp_s = 20
f4 flux_obs0.ki_s = 2
f4 angle0.slip = flux_obs0.slip
f4 ramp0.vel_ext_cmd = 0
at 0.02 f4 ramp0.en = 1
at 0.4 f4 ramp0.vel_ext_cmd = 600
at 8.0 plant load=-3
at 16.0 plant load=0
at 20.0 f4 ramp0.vel_ext_cmd = 0
trace plant.vel flux_obs0.k flux_obs0.pos_err flux_obs0.slip acim_flux0.slip hv0.iq_fb plant.th_flux
expect max fault0.last_fault < 0.5 from 0.5
expect final flux_obs0.k > 0.74 from 15.9 to 16.0
expect final flux_obs0.k < 0.80 from 15.9 to 16.0
expect absmax flux_obs0.pos_err < 0.03 from 12 to 16
run 28
